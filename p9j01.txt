function weekDayList(ws){
 const out=[], d=new Date(ws+"T00:00:00");
 for(let i=0;i<7;i++){ out.push(d.toISOString().slice(0,10)); d.setDate(d.getDate()+1); }
 return out;
}
function weekRangeLabel(ws){
 const days=weekDayList(ws);
 const a=new Date(days[0]+"T00:00:00"), b=new Date(days[6]+"T00:00:00");
 const loc=DB.lang==="de"?"de-DE":"es-AR";
 const fa=a.toLocaleDateString(loc,{day:"numeric",month:"short"});
 const fb=b.toLocaleDateString(loc,{day:"numeric",month:"short"});
 return fa+" – "+fb;
