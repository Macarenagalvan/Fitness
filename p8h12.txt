  const k=order[i];
  if(k==="running"&&run&&!run.done){
   return {load:L,kind:"running",title:t("hoy_type_run"),sub:t("hoy_sub_run",{w:run.week,n:run.session&&run.session.id==="s2"?2:run.session&&run.session.id==="s3"?3:1}),recKey:"hoy_rec_run",ctaKey:"hoy_cta_start",alts:hoyAlts(L,run,yg,stNeed,["running"])};
  }
  if(k==="yoga"&&yg&&!yg.done){
   return {load:L,kind:"yoga",title:t("hoy_type_yg"),sub:t("hoy_sub_yg",{w:yg.week,n:yg.session&&yg.session.id==="s2"?2:1}),recKey:"hoy_rec_yoga",ctaKey:"hoy_cta_start",alts:hoyAlts(L,run,yg,stNeed,["yoga"])};
  }
  if(k==="strength"&&stNeed){
   const nm=st.day?dn(st.day).split("·")[0].trim():t("hoy_type_str");
   return {load:L,kind:"strength",title:nm,sub:t("hoy_sub_str"),recKey:"hoy_rec_str",ctaKey:"hoy_cta_start",alts:hoyAlts(L,run,yg,stNeed,["strength"])};
  }
 }
 return {load:L,kind:"mobility",presetId:"general10",title:t("hoy_type_mob"),sub:t("mob_gen10"),recKey:"hoy_rec_free",ctaKey:"hoy_cta_mob",alts:[{kind:"hub"}]};
}
