function mobFinish(){
 if(!mobLive) return;
 const p=mobPreset(mobLive.presetId);
 const elapsed=Math.round((Date.now()-mobLive.startedAt)/60000);
 const dur=Math.max(1, elapsed|| (p&&p.duration) ||1);
 const entry={type:"mobility",date:today(),presetId:mobLive.presetId,durMin:dur,completed:true,exercises:[]};
 if(!Array.isArray(DB.logs)) DB.logs=[];
 DB.logs.push(entry);
 save();
