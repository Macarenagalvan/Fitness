function mobDose(st){
 if(!st) return "";
 const side=mobSideLabel(st.side);
 let core="";
 if(st.mode==="time") core=t("mob_dose_sec",{n:st.sec});
 else if(st.mode==="breaths") core=t("mob_dose_breaths",{n:st.breaths});
 else if(st.mode==="reps") core=t("mob_dose_reps",{n:st.reps});
 return side?core+" · "+side:core;
}
