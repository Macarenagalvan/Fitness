function nextYogaSession(){
 ensureYogaDB(DB);
 const p=yogaProg(), w=yogaWeekDef(p.week);
 const sess=(w.sessions||[]).filter(s=>!s.optional);
 const pending=sess.find(s=>!yogaIsDone(p.week,s.id));
 if(pending) return {week:p.week,session:pending,done:false};
 return {week:p.week,session:sess[0]||null,done:true};
}
